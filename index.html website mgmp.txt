    <script>
        // GANTI URL INI dengan URL Web App dari Google Apps Script Anda
        const SCRIPT_URL = "https://script.google.com/macros/s/YOUR_SCRIPT_ID/exec";
        let isLogged = false;

        // Inisialisasi Ikon
        lucide.createIcons();

        // Navigasi Halaman
        function showPage(pageId) {
            // Sembunyikan semua halaman
            const pages = ['home', 'berita', 'administrasi', 'galeri', 'anggota', 'admin'];
            pages.forEach(p => {
                document.getElementById('page-' + p).classList.add('d-none');
            });
            
            // Tampilkan halaman terpilih
            document.getElementById('page-' + pageId).classList.remove('d-none');
            window.scrollTo(0, 0);

            // Fetch data jika masuk halaman tertentu
            if (pageId === 'berita') loadBerita();
            if (pageId === 'anggota') loadAnggota();
            if (pageId === 'admin' && isLogged) loadAdminNews();
        }

        // FUNGSI API
        async function fetchData(action) {
            showLoading(true);
            try {
                const response = await fetch(`${SCRIPT_URL}?action=${action}`);
                const data = await response.json();
                showLoading(false);
                return data;
            } catch (err) {
                console.error(err);
                showLoading(false);
                return [];
            }
        }

        // LOAD DATA
        async function loadBerita() {
            const list = document.getElementById('berita-list');
            list.innerHTML = '<div class="col-12 text-center p-5">Memuat berita...</div>';
            
            const data = await fetchData('getBerita');
            list.innerHTML = '';
            
            if (data.length === 0) {
                list.innerHTML = '<div class="col-12 text-center">Belum ada berita.</div>';
                return;
            }

            data.reverse().forEach(item => {
                const col = document.createElement('div');
                col.className = 'col-md-4 mb-4';
                col.innerHTML = `
                    <div class="card h-100 p-3">
                        <small class="text-muted">${item.tanggal}</small>
                        <h5 class="mt-2">${item.judul}</h5>
                        <p class="text-truncate">${item.konten}</p>
                        <button class="btn btn-link text-primary p-0 text-start">Baca Selengkapnya</button>
                    </div>
                `;
                list.appendChild(col);
            });
        }

        async function loadAnggota() {
            const list = document.getElementById('anggota-list');
            list.innerHTML = '<tr><td colspan="2" class="text-center">Memuat data...</td></tr>';
            
            const data = await fetchData('getAnggota');
            list.innerHTML = '';
            
            data.forEach(item => {
                const tr = document.createElement('tr');
                tr.innerHTML = `
                    <td>${item.nama}</td>
                    <td>${item.sekolah}</td>
                `;
                list.appendChild(tr);
            });
        }

        // ADMIN LOGIC
        function checkLogin() {
            const pass = document.getElementById('admin-pass').value;
            // Password dummy: informatika2024
            if (pass === 'informatika2024') {
                isLogged = true;
                document.getElementById('admin-login').classList.add('d-none');
                document.getElementById('admin-panel').classList.remove('d-none');
                loadAdminNews();
            } else {
                alert('Password salah!');
            }
        }

        function logoutAdmin() {
            isLogged = false;
            document.getElementById('admin-pass').value = '';
            document.getElementById('admin-login').classList.remove('d-none');
            document.getElementById('admin-panel').classList.add('d-none');
        }

        async function submitBerita() {
            const judul = document.getElementById('news-title').value;
            const konten = document.getElementById('news-content').value;

            if (!judul || !konten) return alert('Lengkapi data!');

            showLoading(true);
            try {
                await fetch(SCRIPT_URL, {
                    method: 'POST',
                    mode: 'no-cors', // Penting untuk Apps Script POST
                    body: JSON.stringify({ action: 'addBerita', judul, konten })
                });
                alert('Berita berhasil ditambahkan (mungkin butuh beberapa saat untuk muncul)');
                document.getElementById('news-title').value = '';
                document.getElementById('news-content').value = '';
                loadAdminNews();
            } catch (err) {
                console.error(err);
            }
            showLoading(false);
        }

        async function deleteBerita(id) {
            if (!confirm('Hapus berita ini?')) return;
            showLoading(true);
            try {
                await fetch(SCRIPT_URL, {
                    method: 'POST',
                    mode: 'no-cors',
                    body: JSON.stringify({ action: 'deleteBerita', id })
                });
                alert('Permintaan hapus dikirim.');
                loadAdminNews();
            } catch (err) {
                console.error(err);
            }
            showLoading(false);
        }

        async function loadAdminNews() {
            const container = document.getElementById('admin-news-list');
            container.innerHTML = 'Memuat...';
            const data = await fetchData('getBerita');
            container.innerHTML = '';
            data.reverse().forEach(item => {
                const div = document.createElement('div');
                div.className = 'list-group-item d-flex justify-content-between align-items-center';
                div.innerHTML = `
                    <span>${item.judul}</span>
                    <button class="btn btn-sm btn-danger" onclick="deleteBerita('${item.id}')">Hapus</button>
                `;
                container.appendChild(div);
            });
        }

        function showLoading(show) {
            document.getElementById('loading-overlay').style.display = show ? 'flex' : 'none';
        }
